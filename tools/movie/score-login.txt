// The login screen's music: calm, warm, a little mysterious. D major at 75 beats a minute, twenty bars of 3.2 s (64 s), written to loop: the renderer
// plays four more seconds and folds them back over the start, so the end runs into the beginning without a seam.
// Strings and a low drone hold the ground; a piano arpeggio walks underneath; a bell melody, a soft choir and a plucked zither take turns on top.
const BAR = 3.2, B8 = BAR / 8, E0 = m => 440 * Math.pow(2, (m - 69) / 12);
const PBUF = {};
function plBuf(m) { if (PBUF[m]) return PBUF[m]; const f = E0(m), N = Math.max(2, Math.round(SR / f)), len = SR * 3, b = ac.createBuffer(1, len, SR), d = b.getChannelData(0), ring = new Float32Array(N); for (let i = 0; i < N; i++) ring[i] = (rnd() * 2 - 1) * (i / N < .5 ? 1 : .55); let p = 0, damp = .4985 + Math.min(.0012, f / 900000); for (let t = 0; t < len; t++) { const n = (p + 1) % N, y = ring[p]; ring[p] = (y + ring[n]) * damp; d[t] = y; p = n; } return PBUF[m] = b; }
function pluck(t, m, v = .5, dur = 1.8, p = -.2, bend = 0) { const s = ac.createBufferSource(); s.buffer = plBuf(m); const g = G(0); g.gain.setValueAtTime(v * .8, t); g.gain.setTargetAtTime(0, t + dur, .3); if (bend) { s.detune.setValueAtTime(0, t + .15); for (let k = 0; k < dur * 6; k++) s.detune.linearRampToValueAtTime((k % 2 ? -1 : 1) * bend * Math.min(1, k / 4), t + .15 + (k + 1) * .17); } s.connect(g); pan(g, p, PIANO); s.start(t); s.stop(t + dur + 1.2); }
// D major: D Bm G A | D F#m G A, voiced warm and open
const D_ = [38, 57, 62, 66], Bm = [35, 54, 59, 62], G_ = [31, 55, 59, 62], A_ = [33, 52, 57, 61], Fm = [30, 54, 57, 61];
const PROG = [D_, Bm, G_, A_, D_, Fm, G_, A_];
const NB = Math.ceil(TOTAL / BAR);
for (let b = 0; b < NB; b++) {
  const t = b * BAR, c = PROG[b % 8], sec = Math.floor(b / 4) % 5;
  chord(t, c.slice(1).map(m => m + 12), BAR * 1.02, .17 + (sec >= 2 ? .04 : 0), { bright: .22, a: .9, r: 1.2 });
  lowStr(t, c[0], BAR, .22, 520);
  // the piano walks the chord in eighths, softly, every bar
  const arp = [c[0] + 12, c[1], c[2], c[3], c[2] + 12, c[3], c[2], c[1]];
  arp.forEach((m, i) => piano(t + i * B8, m + 12, .16 + (i == 0 ? .06 : 0), B8 * 2.2, (i % 2 ? .2 : -.2)));
}
drone(0, TOTAL, [26, 38, 45], .16, 200);
// the bell melody: bars 4-11, a slow rising line that answers itself
const MEL = [[0, 78, 1.5], [1.5, 76, .5], [2, 74, 1], [3, 76, 1], [4, 81, 2], [6, 78, 1], [7, 76, 1], [8, 74, 1.5], [9.5, 73, .5], [10, 71, 2], [12, 73, 1], [13, 74, 3]];
for (const rep of [0, 1]) MEL.forEach(([bt, m, d]) => { const t = (4 + rep * 4) * BAR + bt * (BAR / 4); bell(t, m, .2 - rep * .03, .15); strings(t, m - 12, d * BAR / 4, .1, {}); });
// the choir breathes in under bars 8-15
for (let b = 8; b < 16; b++) { const c = PROG[b % 8]; choirs(b * BAR + .2, [c[1] + 12, c[2] + 12, c[3] + 12], BAR * .98, .1, 'a', {}); }
// the zither answers in bars 12-19, with a little waver on the held notes
const ZM = [[0, 74, 1], [1, 78, 1], [2, 81, 2], [4, 83, 1], [5, 81, 1], [6, 78, 2], [8, 76, 1], [9, 78, 1], [10, 74, 3], [13, 71, 1], [14, 73, 2]];
for (const rep of [0, 1]) ZM.forEach(([bt, m, d], i) => { const t = (12 + rep * 4) * BAR + bt * (BAR / 4); pluck(t, m - 12, .38, d * BAR / 4 + .6, -.1 + (i % 3) * .1, d > 1.5 ? 16 : 0); });
// air: wind under everything, a shimmer every eight bars
SFX.wind(0, TOTAL);
for (let b = 0; b < NB; b += 8) SFX.shimmer(b * BAR + 1.6, .45);
